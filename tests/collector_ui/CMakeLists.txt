cmake_minimum_required(VERSION 3.20)
project(collector_ui_host C)
set(CMAKE_C_STANDARD 11)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(COLLECTOR_UI_HEAP_KIB 24 CACHE STRING "LVGL host test pool in KiB")
add_subdirectory("${LVGL_SOURCE}" lvgl)
target_compile_definitions(lvgl PUBLIC COLLECTOR_UI_HEAP_KIB=${COLLECTOR_UI_HEAP_KIB})
add_executable(collector_ui_host test_collector_ui.c
    ../../assets/fonts/collector_font_16.c
    ../../components/bsp/src/bsp_display_rounding.c)
target_include_directories(collector_ui_host PRIVATE stubs ../../main ../../components/bsp/include
    ../../components/bsp/src "${CMAKE_BINARY_DIR}")
if(MSVC)
    target_compile_options(collector_ui_host PRIVATE /W4 /WX /utf-8)
else()
    target_compile_options(collector_ui_host PRIVATE -Wall -Wextra -Werror)
endif()
target_link_libraries(collector_ui_host PRIVATE lvgl)
if(UNIX)
    target_link_libraries(collector_ui_host PRIVATE m)
endif()
